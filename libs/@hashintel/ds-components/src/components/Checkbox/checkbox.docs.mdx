export const ResolvesIndeterminate = ({ children }) => {
  const child = Children.only(children);
  const [state, setState] = useState({
    indeterminate: !!child.props.indeterminate,
    value: !!child.props.value,
  });
  return cloneElement(child, {
    ...state,
    onChange: (value) => setState({ indeterminate: false, value }),
  });
};

## Installation

1. Install the package.

```sh
npm install @hashintel/ds-components
```

2. Import the component.

```tsx
import { useState } from "react";

import { Checkbox } from "@hashintel/ds-components";

export const IncludeArchivedOption = () => {
  const [includeArchived, setIncludeArchived] = useState(false);

  return (
    <Checkbox
      label="Include archived entities"
      value={includeArchived}
      onChange={setIncludeArchived}
    />
  );
};
```

### Indeterminate

Use `indeterminate` on a parent checkbox when only some of its children are selected. It's display-only — `value` remains true or false.

export const selectAllCode = `import { useState } from "react";

import { Checkbox } from "@hashintel/ds-components";

const properties = ["name", "website"];

export const PropertySelection = () => {
  const [selected, setSelected] = useState(["name"]);
  const allSelected = selected.length === properties.length;

  const toggle = (property: string) => (checked: boolean) =>
    setSelected(
      checked
        ? [...selected, property]
        : selected.filter((entry) => entry !== property),
    );

  return (
    <>
      <Checkbox
        indeterminate={selected.length > 0 && !allSelected}
        value={allSelected}
        label="All properties"
        onChange={(checked) => setSelected(checked ? properties : [])}
      />
      <Checkbox
        value={selected.includes("name")}
        label="Name"
        onChange={toggle("name")}
      />
      <Checkbox
        value={selected.includes("website")}
        label="Website"
        onChange={toggle("website")}
      />
    </>
  );
};`;

### In forms

`value` is the checked boolean, not what the form submits: when checked, the hidden native input submits `htmlValue` (default `"on"`) under `name`, and nothing when unchecked. `required` blocks native submission until checked.

```tsx
import { useState } from "react";

import { Button, Checkbox } from "@hashintel/ds-components";

export const InviteTeammateForm = () => {
  const [notifyByEmail, setNotifyByEmail] = useState(true);
  const [dpaAccepted, setDpaAccepted] = useState(false);

  return (
    <form action="/api/invites" method="post">
      <Checkbox
        label="Notify the teammate by email"
        name="notify"
        htmlValue="email"
        value={notifyByEmail}
        onChange={setNotifyByEmail}
      />
      <Checkbox
        label="I agree to the data processing agreement"
        name="dpa"
        required
        value={dpaAccepted}
        onChange={setDpaAccepted}
      />
      <Button type="submit" tone="brand">
        Send invite
      </Button>
    </form>
  );
};
```

## Behavior

- Checkbox renders as one `<label>` wrapping a hidden native checkbox, so clicking anywhere on it toggles. Don't nest it inside another label or interactive element.
- Long labels wrap rather than truncate. If you prefer truncation, apply it with a `className`.
- `invalid` sets `aria-invalid` on the input.

## Content

- Phrase labels positively: "Notify teammates by email", never "Don't send notifications" — unchecking a negative is a double negative.
- Lead with the outcome of checking, in sentence case with no terminal punctuation: "Include archived entities", "Sync to GitHub".
- Try to keep labels to one line — under roughly 8 words. It's generally better to move caveats and detail into nearby helper text.

## Accessibility

Checkbox implements the [WAI-ARIA checkbox pattern](https://www.w3.org/WAI/ARIA/apg/patterns/checkbox/).

### What you must do

- Always provide `label` — without it the input has no accessible name. When the visible label lives elsewhere, pass `htmlForId` and point that label's `htmlFor` at it.
- Don't signal errors with the red box alone — put the error text next to the checkbox.
- Prefer explaining a constraint over `disabled` — a disabled box has reduced contrast and doesn't say why it's locked.
