fn out_dir() -> PathBuf {
    let dir = std::env::var("VERIFY_OUT")
        .map(PathBuf::from)
        .unwrap_or_else(|_| std::env::temp_dir().join("verify_specs"));
    std::fs::create_dir_all(&dir).expect("out dir");
    dir
}

fn cell(value: &Value) -> String {
    match value {
        Value::Text(text) => text.clone(),
        Value::Boolean(flag) => flag.to_string(),
        Value::Null => "null".to_owned(),
        Value::BigInt(n) => n.to_string(),
        Value::UBigInt(n) => n.to_string(),
        Value::Int(n) => n.to_string(),
        other => format!("{other:?}"),
    }
}

async fn lines(workspace: &Workspace, sql: &str) -> Vec<String> {
    let rows = workspace
        .query(sql.into())
        .await
        .unwrap_or_else(|report| panic!("query failed: {sql}\n{report:?}"));
    rows.iter()
        .map(|row| row.iter().map(cell).collect::<Vec<_>>().join("|"))
        .collect()
}

async fn variants(workspace: &Workspace, sql: &str) -> Vec<String> {
    let rows = workspace.query(sql.into()).await.expect("query");
    rows.iter()
        .next()
        .map(|row| row.iter().map(|value| format!("{value:?}")).collect())
        .unwrap_or_default()
}

async fn create(workspace: &Workspace, table: &str, sql: &str) {
    workspace
        .create_table(table.into(), sql.into())
        .await
        .unwrap_or_else(|report| panic!("create_table {table} failed: {sql}\n{report:?}"));
}

#[tokio::test]
async fn a1_rowid() {
    let directory = TempDir::new().expect("dir");
    let workspace = open(&directory).await;
    create(
        &workspace,
        "_step/t",
        "SELECT i AS id, 'v' || i AS col FROM range(0, 300000) AS r(i)",
    )
    .await;
    let stats = lines(
        &workspace,
        "SELECT count(*), count(DISTINCT rowid), min(rowid), max(rowid) FROM \"_step/t\"",
    )
    .await;
    println!("A1 count, distinct rowid, min, max: {stats:?}");
    let keys = variants(
        &workspace,
        r#"SELECT count(*) AS "rows", min(rowid) AS "first", max(rowid) AS "last" FROM "_step/t""#,
    )
    .await;
    println!("A1 row_keys variants: {keys:?}");

    let first = lines(&workspace, "SELECT rowid, id FROM \"_step/t\" ORDER BY rowid").await;
    create(&workspace, "other", "SELECT 1 AS x").await;
    let _ = lines(&workspace, "SELECT count(*) FROM \"_step/t\" WHERE id % 7 = 0").await;
    let second = lines(&workspace, "SELECT rowid, id FROM \"_step/t\" ORDER BY rowid").await;
    println!(
        "A1 rowid->id mapping equal across queries: {}",
        first == second
    );
    assert_eq!(first, second);

    let mut total = 0_usize;
    let mut start = 0_u64;
    while start < 300_000 {
        let end = start + 65_536;
        let rows = lines(
            &workspace,
            &format!(
                "SELECT rowid, col FROM \"_step/t\" WHERE rowid >= {start} AND rowid < {end} ORDER BY rowid"
            ),
        )
        .await;
        total += rows.len();
        start = end;
    }
    println!("A1 range reads total rows: {total}");
    assert_eq!(total, 300_000);
    let range_variant = variants(
        &workspace,
        "SELECT rowid AS \"_row\", col AS \"id\" FROM \"_step/t\" WHERE rowid >= 0 AND rowid < 2",
    )
    .await;
    println!("A1 read_range variants: {range_variant:?}");

    // A table whose source has a column named rowid.
    let created = workspace
        .create_table(
            "_step/r".into(),
            "SELECT * FROM (VALUES (7, 'x'), (9, 'y')) AS v(rowid, col)".into(),
        )
        .await;
    println!("A1 create_table with column rowid: {created:?}");
    if created.is_ok() {
        println!(
            "A1 SELECT rowid FROM table with rowid column: {:?}",
            lines(&workspace, "SELECT rowid, col FROM \"_step/r\" ORDER BY col").await
        );
        println!(
            "A1 SELECT * of it: {:?}",
            lines(&workspace, "SELECT * FROM \"_step/r\" ORDER BY col").await
        );
        println!(
            "A1 range query on it rowid >= 0 AND rowid < 2: {:?}",
            lines(
                &workspace,
                "SELECT rowid, col FROM \"_step/r\" WHERE rowid >= 0 AND rowid < 2"
            )
            .await
        );
        let hashed = workspace
            .create_table(
                "_step/r2".into(),
                "SELECT source.*, md5('') AS _content_hash FROM \"_step/r\" AS source".into(),
            )
            .await;
        println!("A1 SELECT source.* over a rowid column: {hashed:?}");
    }
    println!(
        "A1 SELECT * of table without rowid column returns columns: {:?}",
        workspace
            .query("SELECT * FROM \"_step/t\" LIMIT 1".into())
            .await
            .expect("query")
            .columns()
    );
}

const LABEL_SQL: &str = r#"WITH prior AS (
    SELECT entity_id, kind, target, source_id, content_hash, live, TRUE AS in_base
    FROM "_diff/base"
    UNION ALL
    SELECT entity_id, kind, target, source_id, content_hash, FALSE AS live, FALSE AS in_base
    FROM "_diff/carried"
),
joined AS (
    SELECT
        coalesce(r.entity_id, p.entity_id) AS entity_id,
        coalesce(r.kind, p.kind) AS kind,
        coalesce(r.target, p.target) AS target,
        coalesce(r.source_id, p.source_id) AS source_id,
        coalesce(r.content_hash, p.content_hash) AS content_hash,
        r.entity_id IS NOT NULL AS present,
        coalesce(p.in_base, FALSE) AS in_base,
        coalesce(p.live, FALSE) AS was_live,
        coalesce(r.content_hash = p.content_hash, FALSE) AS same_hash,
        s.entity_id IS NOT NULL AS resend,
        {protected} AS protected
    FROM "_diff/rows" AS r
    FULL OUTER JOIN prior AS p ON p.entity_id = r.entity_id
    LEFT JOIN "_diff/resend" AS s ON s.entity_id = coalesce(r.entity_id, p.entity_id)
)
SELECT
    entity_id,
    kind,
    target,
    source_id,
    content_hash,
    present OR (was_live AND protected AND NOT resend) AS live,
    CASE
        WHEN present AND NOT in_base THEN 'insert'
        WHEN present AND NOT was_live THEN 'update'
        WHEN present AND NOT same_hash THEN 'update'
        WHEN present AND (resend OR {resend_all}) THEN 'resend'
        WHEN present THEN NULL
        WHEN NOT in_base THEN 'archive'
        WHEN was_live AND resend THEN 'archive'
        WHEN was_live AND protected THEN NULL
        WHEN was_live THEN 'archive'
        WHEN resend OR {resend_all} THEN 'archive'
        ELSE NULL
    END AS change,
    present,
    was_live,
    resend
FROM joined"#;

fn label_sql(protected: &str, resend_all: bool) -> String {

fn compare(name: &str, actual: &[String], expected: &[String]) -> bool {
    let equal = actual == expected;
    println!("A3 {name}: {}", if equal { "MATCH" } else { "MISMATCH" });
    if !equal {
        for (index, line) in actual.iter().enumerate() {
            let want = expected.get(index).map(String::as_str).unwrap_or("<none>");
            if line != want {
                println!("   got  {line}\n   want {want}");
            }
        }
        if expected.len() != actual.len() {
            println!("   lengths {} vs {}", actual.len(), expected.len());
        }
    }
    equal
}

fn patch(expected: &[&str], changes: &[(&str, &str)]) -> Vec<String> {
    expected
        .iter()
        .map(|line| {
            let id = &line[..3];
            changes
                .iter()
                .find(|(key, _)| *key == id)
                .map_or_else(|| (*line).to_owned(), |(_, new)| (*new).to_owned())
        })
        .collect()
}

#[tokio::test]
async fn a3_resend_carried_union_duplicates_changes() {
    let directory = TempDir::new().expect("dir");
    let workspace = open(&directory).await;

    // empty_sql forms
    for (name, sql) in [("empty hash", EMPTY_HASH), ("empty row", EMPTY_ROW), ("empty id", EMPTY_ID)] {
        let result = workspace.create_table(format!("_e/{name}").into(), sql.into()).await;
        println!("A3 {name}: {result:?}");
        assert!(result.is_ok());
    }

    // resend_sql and carried_sql for two listed runs and a rejected file
    create(&workspace, "_diff/sending/0/writes", "SELECT * FROM (VALUES ('a'), ('b'), ('c')) AS t(entity_id)").await;
    create(&workspace, "_diff/sending/1/writes", "SELECT * FROM (VALUES ('b'), ('d')) AS t(entity_id)").await;
    create(&workspace, "_diff/rejected", "SELECT * FROM (VALUES ('x'), ('a')) AS t(entity_id)").await;
    create(&workspace, "_diff/sending/0/hashes", "SELECT * FROM (VALUES ('a','entity','t','sa','h0a',true), ('b','entity','t','sb','h0b',true), ('c','entity','t','sc','h0c',true), ('z','entity','t','sz','h0z',true)) AS t(entity_id, kind, target, source_id, content_hash, live)").await;
    create(&workspace, "_diff/sending/1/hashes", "SELECT * FROM (VALUES ('b','entity','t','sb','h1b',true), ('d','link','l','sd','h1d',true)) AS t(entity_id, kind, target, source_id, content_hash, live)").await;
    create(&workspace, "_diff/base", "SELECT * FROM (VALUES ('c','entity','t','sc','h0c',true)) AS t(entity_id, kind, target, source_id, content_hash, live)").await;

    let resend_sql = r#"SELECT DISTINCT entity_id
FROM (
    SELECT entity_id FROM "_diff/sending/0/writes"
    UNION ALL
    SELECT entity_id FROM "_diff/sending/1/writes"
    UNION ALL
    SELECT entity_id FROM "_diff/rejected"
) AS listed"#;
    create(&workspace, "_diff/resend", resend_sql).await;
    let resend = lines(&workspace, "SELECT entity_id FROM \"_diff/resend\" ORDER BY entity_id").await;
    println!("A3 resend_sql: {resend:?}");

fn write_ids(path: &Path, rows: &[(u64, Option<String>, Option<String>)]) {
    let file = File::create(path).expect("file");
    let properties = WriterProperties::builder()
        .set_compression(Compression::SNAPPY)
        .set_max_row_group_row_count(Some(64 * 1024))
        .build();
    let batch = |chunk: &[(u64, Option<String>, Option<String>)]| {
        RecordBatch::try_from_iter_with_nullable(vec![
            (
                "_row",
                Arc::new(UInt64Array::from(chunk.iter().map(|r| r.0).collect::<Vec<_>>())) as ArrayRef,
                false,
            ),
            (
                "_entity_id",
                Arc::new(StringArray::from(chunk.iter().map(|r| r.1.clone()).collect::<Vec<_>>())) as ArrayRef,
                true,
            ),
            (
                "_source_id",
                Arc::new(StringArray::from(chunk.iter().map(|r| r.2.clone()).collect::<Vec<_>>())) as ArrayRef,
                true,
            ),
        ])
        .expect("batch")
    };
    let schema = batch(&[]).schema();
    let mut writer = ArrowWriter::try_new(file, schema, Some(properties)).expect("writer");
    for chunk in rows.chunks(1024) {
        writer.write(&batch(chunk)).expect("write");
    }
    writer.close().expect("close");
}

fn sha256(path: &Path) -> String {
    let output = std::process::Command::new("shasum")
        .args(["-a", "256"])
        .arg(path)
        .output()
        .expect("shasum");
    String::from_utf8_lossy(&output.stdout)
        .split_whitespace()
        .next()
        .unwrap_or_default()
        .to_owned()
}

fn describe(path: &Path) {
    let reader = SerializedFileReader::new(File::open(path).expect("file")).expect("reader");
    let metadata = reader.metadata().file_metadata();
    println!(
        "A6   {}: created_by {:?}, key_value keys {:?}, row groups {}",
        path.file_name().unwrap().to_string_lossy(),
        metadata.created_by(),
        metadata
            .key_value_metadata()
            .map(|kv| kv.iter().map(|entry| entry.key.clone()).collect::<Vec<_>>()),
        reader.metadata().num_row_groups()
    );
}

fn write_rowfile(path: &Path, count: u64) {
    let file = File::create(path).expect("file");
    let properties = WriterProperties::builder()
        .set_compression(Compression::SNAPPY)
        .set_max_row_group_row_count(Some(65_536))
        .build();
    let rows: Vec<u64> = (0..count).collect();
    let batch = |chunk: &[u64]| {
        RecordBatch::try_from_iter_with_nullable(vec![
            (
                "entity_id",
                Arc::new(StringArray::from(
                    chunk.iter().map(|i| if i % 13 == 0 { None } else { Some(format!("web~{i:08}")) }).collect::<Vec<_>>(),
                )) as ArrayRef,
                true,
            ),
            ("n", Arc::new(UInt64Array::from(chunk.to_vec())) as ArrayRef, false),
            (
                "flag",
                Arc::new(BooleanArray::from(chunk.iter().map(|i| i % 2 == 0).collect::<Vec<_>>())) as ArrayRef,
                false,
            ),
        ])
        .expect("batch")
    };
    let schema = batch(&[]).schema();
    let mut writer = ArrowWriter::try_new(file, schema, Some(properties)).expect("writer");
    for chunk in rows.chunks(1024) {
        writer.write(&batch(chunk)).expect("write");
        if writer.in_progress_size() >= 8 * 1024 * 1024 {
            writer.flush().expect("flush");
        }
    }
    writer.close().expect("close");
}

    write_rowfile(&out.join("rowfile-empty-1.parquet"), 0);
    write_rowfile(&out.join("rowfile-empty-2.parquet"), 0);

        "A6 empty RowFile identical: {}",

    );

        .load_capture(SourceCapture::open(out.join("rowfile-1.parquet")).expect("open"), "rf".into())
        .await
        .expect("load");
    println!(
        "A6 RowFile columns loaded: {:?}",
        lines(&workspace, "SELECT column_name || ':' || data_type FROM information_schema.columns WHERE table_name = 'rf' ORDER BY ordinal_position").await
    );
    assert!(a == b && x == y);
}

async fn a7_type_round_trip() {
    let directory = TempDir::new().expect("dir");
    let workspace = open(&directory).await;
    let columns = [
        ("c_varchar", "VARCHAR", "'text'"),
        ("c_bigint", "BIGINT", "-9007199254740993"),
        ("c_hugeint", "HUGEINT", "12345678901234567890123456789"),
        ("c_decimal", "DECIMAL(18,2)", "12345.67"),
        ("c_double", "DOUBLE", "0.1"),
        ("c_boolean", "BOOLEAN", "true"),
        ("c_date", "DATE", "DATE '2026-01-02'"),
        ("c_timestamp", "TIMESTAMP", "TIMESTAMP '2026-01-02 03:04:05.123456'"),
        ("c_timestamptz", "TIMESTAMPTZ", "TIMESTAMPTZ '2026-01-02 03:04:05.123456+00'"),
    ];
    let row = |null: bool| {
        columns
            .iter()
            .map(|(name, ty, value)| {
                format!("CAST({} AS {ty}) AS {name}", if null { "NULL" } else { value })
            })
            .collect::<Vec<_>>()
            .join(", ")
    };
    create(&workspace, "types_src", &format!("SELECT 0 AS k, {} UNION ALL SELECT 1 AS k, {}", row(false), row(true))).await;
    let types = |table: &str| {
        format!("SELECT column_name, data_type FROM information_schema.columns WHERE table_name = '{table}' ORDER BY ordinal_position")
    };
    let before = lines(&workspace, &types("types_src")).await;
    let capture = workspace
        .capture("SELECT * FROM types_src ORDER BY k".into(), directory.path().join("types.parquet"))
        .await
        .expect("capture");
    let reader = SerializedFileReader::new(File::open(capture.path()).expect("f")).expect("r");
    let arrow_schema = parquet::arrow::parquet_to_arrow_schema(
        reader.metadata().file_metadata().schema_descr(),
        reader.metadata().file_metadata().key_value_metadata(),
    )
    .expect("schema");
    workspace.load_capture(capture, "types_back".into()).await.expect("load");
    let after = lines(&workspace, &types("types_back")).await;
    for index in 0..before.len() {
        let field = arrow_schema
            .fields()
            .get(index)
            .map(|f| format!("{:?}", f.data_type()))
            .unwrap_or_default();
        println!(
            "A7 {:<26} -> {:<32} arrow {}",
            before[index],
            after.get(index).cloned().unwrap_or_default(),
            field
        );
    }
    for (name, _, _) in columns {
        for (label, table) in [("source", "types_src"), ("loaded", "types_back")] {
            let result = workspace
                .query(format!("SELECT {name} FROM {table} ORDER BY k").into())
                .await;
            match result {
                Ok(rows) => println!(
                    "A7 query {name} ({label}): {:?}",
                    rows.iter().map(|r| format!("{:?}", r[0])).collect::<Vec<_>>()
                ),
                Err(report) => println!("A7 query {name} ({label}): ERROR {:?}", report.current_context()),
            }
        }
    }
    let equal = lines(&workspace, "SELECT count(*) FROM types_src s JOIN types_back b USING (k) WHERE s.c_varchar IS NOT DISTINCT FROM b.c_varchar AND s.c_bigint IS NOT DISTINCT FROM b.c_bigint AND s.c_hugeint IS NOT DISTINCT FROM b.c_hugeint AND s.c_decimal IS NOT DISTINCT FROM b.c_decimal AND s.c_double IS NOT DISTINCT FROM b.c_double AND s.c_boolean IS NOT DISTINCT FROM b.c_boolean AND s.c_date IS NOT DISTINCT FROM b.c_date AND s.c_timestamp IS NOT DISTINCT FROM b.c_timestamp AND s.c_timestamptz IS NOT DISTINCT FROM b.c_timestamptz").await;
    println!("A7 rows whose values compare equal after round trip (of 2): {equal:?}");
}

const NAMESPACE_HEX: &str = "d6e2c7a1f84b4e3a9c0d5b7f1e3a2d4c";

fn uuid_sql(name: &str) -> String {
    // `h` is the 40-digit hex text of the SHA-1.
    let h = format!("sha1(from_hex('{NAMESPACE_HEX}') || encode({name}))");
    format!(
        "substr({h}, 1, 8) || '-' || substr({h}, 9, 4) || '-5' || substr({h}, 14, 3) || '-' || substr('89ab89ab89ab89ab', position(substr({h}, 17, 1) IN '0123456789abcdef'), 1) || substr({h}, 18, 3) || '-' || substr({h}, 21, 12)"
    )
}

#[tokio::test]
async fn a8_entity_id_in_sql() {
    let directory = TempDir::new().expect("dir");
    let workspace = open(&directory).await;
    for probe in [
        "SELECT sha1(from_hex('00ff'))",
        "SELECT typeof(from_hex('00ff') || encode('a'))",
        "SELECT sha1('abc')",
        "SELECT sha1(encode('abc'))",
    ] {
        let result = workspace.query(probe.into()).await;
        println!(
            "A8 probe {probe}: {:?}",
            result
                .map(|rows| rows.iter().map(|r| format!("{:?}", r[0])).collect::<Vec<_>>())
                .map_err(|report| format!("{:?}", report.current_context()))
        );
    }
    #[derive(serde::Deserialize)]
    #[serde(rename_all = "camelCase")]
    struct Case {
        ns: String,
        entity_type: String,
        entity_id: String,
        uuid: String,
    }
    let cases: Vec<Case> =
        serde_json::from_str(include_str!("fixtures/entity-ids-typescript.json")).expect("json");
    let web_id = WebId::new(Uuid::from_u128(0xA2C8_E6C7_485D_4B54_A248_53C9_B9B0_14A2));
    let mut names: Vec<(String, String, String, String)> = cases
        .iter()
        .map(|case| {
            (case.ns.clone(), case.entity_type.clone(), case.entity_id.clone(), case.uuid.clone())
        })
        .collect();
    for (ns, ty, id) in [
        ("conn-ü", "https://x/@t/types/entity-type/material/v/1", "Ünïcödé ✓ 𝄞"),
        ("conn-a", "https://x/@t/types/entity-type/material/v/1", "::"),
        ("a::b", "https://x/@t/types/entity-type/material/v/2", "x::y::z"),
        ("conn-a", "https://x/@t/types/entity-type/material/v/1", "trailing space "),
        ("日本", "https://x/@t/types/entity-type/plant/v/1", "工場-01"),
    ] {
        let url: VersionedUrl = ty.parse().expect("url");
        let derived = derive_entity_id(web_id, ns, &url, id);
        names.push((ns.into(), ty.into(), id.into(), derived.entity_uuid.to_string()));
    }
    // Check the fixture UUIDs agree with derive_entity_id too.
    for (ns, ty, id, uuid) in &names {
        let url: VersionedUrl = ty.parse().expect("url");
        assert_eq!(&derive_entity_id(web_id, ns, &url, id).entity_uuid.to_string(), uuid);
    }
    let literal = |text: &str| format!("'{}'", text.replace('\'', "''"));
    let values = names
        .iter()
        .enumerate()
        .map(|(index, (ns, ty, id, _))| {
            format!("({index}, {}, {}, {})", literal(ns), literal(ty), literal(id))
        })
        .collect::<Vec<_>>()
        .join(", ");
    create(&workspace, "names", &format!("SELECT * FROM (VALUES {values}) AS t(i, ns, ty, source_id)")).await;
    let expression = uuid_sql("ns || '::' || ty || '::' || source_id");
    println!("A8 expression: {expression}");
    let sql = format!("SELECT i, {expression} AS uuid FROM names ORDER BY i");
    match workspace.query(sql.into()).await {
        Ok(rows) => {
            let mut matched = 0;
            for (row, (ns, ty, id, want)) in rows.iter().zip(&names) {
                let got = cell(&row[1]);
                let ok = &got == want;
                matched += usize::from(ok);
                println!("A8 {} ns={ns} type={ty} id={id:?} sql={got} rust={want}", if ok { "MATCH" } else { "DIFF " });
            }
            println!("A8 matched {matched} of {}", names.len());
        }
        Err(report) => println!("A8 query failed: {report:?}"),
    }
    // One million rows.
    let million = format!(
        "SELECT i, {} AS uuid FROM range(0, 1000000) AS r(i)",
        uuid_sql("'conn-a::https://x/@t/types/entity-type/material/v/1::' || i")
    );
    let start = Instant::now();
    let result = workspace.create_table("million".into(), million.into()).await;
    println!("A8 one million rows: {:?} in {:?}", result.map(|()| "ok"), start.elapsed());
    let rust_start = Instant::now();
    let url: VersionedUrl = "https://x/@t/types/entity-type/material/v/1".parse().expect("url");
    let mut last = String::new();
    for i in 0..1_000_000_u32 {
        last = derive_entity_id(web_id, "conn-a", &url, &i.to_string()).entity_uuid.to_string();
    }
    println!("A8 Rust derive_entity_id one million (debug build): {:?}", rust_start.elapsed());
    let check = lines(&workspace, "SELECT uuid FROM million WHERE i = 999999").await;
    println!("A8 row 999999 sql {check:?} rust {last}");
}
